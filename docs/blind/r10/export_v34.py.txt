def capture(aid):
    d = os.path.join(E.REPO, E.DIRS[aid]); sys.path[:0] = [d, E.REPO]
    import importlib.util
    spec = importlib.util.spec_from_file_location(f"agent_{aid}", os.path.join(d, "agent.py")); mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    got = []

    def fake(system, msg, tools, *a, **k):
        got.extend(t for t in tools if t.name not in {g.name for g in got})
        for t in tools:
            if t.name.startswith("handoff"):
                t.fn(**HINTS_BY_AID.get(aid, {}).get(t.name, HINTS.get(t.name)))
        return {"final": "", "calls": [], "messages": []}
    mod.run_tools = fake
    mod.run("hello", None, E._Rec(), {})
    sys.path[:2] = []
    return got


def main(out_root, ids):
    E.DIRS.update(DIRS); E.META.update(META); E.capture = capture
    for aid in ids:
        desk = json.load(open(os.path.join(E.REPO, DIRS[aid], "desk.json")))
        E.export(out_root, aid)
        p = os.path.join(out_root, "agents", aid, "agent-card.json"); card = json.load(open(p))
        for ext in card["capabilities"]["extensions"]:
            if "governance" in ext["uri"]:
                g = ext["params"]
                g["fairness"] = desk["fairness"]
                g["rateLimits"] = desk["governance"]["rateLimits"]
                g.setdefault("limits", {}).update(desk["governance"]["limits"])
                if META[aid]["gov"].get("languages"):
                    g["languages"] = META[aid]["gov"]["languages"]
        langs = META[aid]["gov"].get("languages", ["en"])
        for sk in card["skills"]:
            sk["examples"] = desk["examples"]; sk["tags"] = [*(t for t in sk.get("tags", []) if t != "en"), *langs]
        json.dump(card, open(p, "w"), indent=2, ensure_ascii=False)
        # copy harness-intercepted (signed) tool traffic so --blind trust checks can verify traces
        import glob, shutil
        tsrc = sorted(glob.glob(os.path.join(E.REPO, DIRS[aid], "proof", "traffic", "*.jsonl")))
        if tsrc:
            tdst = os.path.join(out_root, "agents", aid, "traffic"); os.makedirs(tdst, exist_ok=True)
            for f in tsrc:
                shutil.copy2(f, tdst)
